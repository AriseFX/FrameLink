cmake_minimum_required(VERSION 3.20)
project(ScreenCastSendCpp VERSION 2.4.4 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ScreenCastSendCpp targets Windows x64; use a Windows toolchain or tools/build_mingw.py")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "NDI ABI and bundled runtime require a 64-bit build")
endif()
enable_language(RC)
add_executable(ScreenCastSendCpp WIN32 src/main.cpp src/selftest.cpp resources/app.rc)
target_include_directories(ScreenCastSendCpp PRIVATE resources)
target_compile_features(ScreenCastSendCpp PRIVATE cxx_std_17)
target_compile_definitions(ScreenCastSendCpp PRIVATE UNICODE _UNICODE NOMINMAX)
file(GLOB YUV_SOURCES third_party/libyuv/source/*.cc)
list(FILTER YUV_SOURCES EXCLUDE REGEX "_(neon|neon64|sve|sme|lsx|lasx)")
add_library(yuv STATIC ${YUV_SOURCES})
target_include_directories(yuv PUBLIC third_party/libyuv/include)
target_link_libraries(ScreenCastSendCpp PRIVATE yuv)
target_link_libraries(ScreenCastSendCpp PRIVATE d3d11 dxgi uuid gdi32 user32 comctl32 shell32 ole32 winmm)
if(MSVC)
  target_compile_options(ScreenCastSendCpp PRIVATE /utf-8 /W4)
  target_compile_options(yuv PRIVATE /utf-8)
  # The application and the reference package both ship as a self-contained EXE.
  set_property(TARGET ScreenCastSendCpp yuv PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(ScreenCastSendCpp PRIVATE -municode -static)
endif()
foreach(runtime_file "Processing.NDI.Lib.x64.dll" "NDI License Agreement.pdf" "Processing.NDI.Lib.Licenses.txt")
  add_custom_command(TARGET ScreenCastSendCpp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/${runtime_file}" "$<TARGET_FILE_DIR:ScreenCastSendCpp>")
endforeach()
